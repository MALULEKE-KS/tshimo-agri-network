! ============================================================
! ISP — simulated upstream provider
! Cisco 2911
! Role: far end of the R1 WAN /30, plus a loopback acting as an
!       "internet" test target for NAT verification.
!
! ISP port facing R1 Gi0/1 is Gi0/0. Confirmed 2 Oct 2026: this file was
! applied unchanged, R1 Gi0/1 came up/up, and a PC reached 203.0.113.1
! through NAT — impossible unless the link lands on the addressed Gi0/0.
! ============================================================

enable
configure terminal
hostname ISP
no ip domain-lookup

interface gigabitEthernet0/0
 description WAN to Tshimo R1 Gi0/1
 ip address 10.23.1.2 255.255.255.252
 no shutdown
exit

! Simulated internet host
interface loopback0
 description Internet test target
 ip address 203.0.113.1 255.255.255.255
exit

end
write memory

! No route back to 10.23.0.0/24 is needed: R1 NATs every inside
! host to 10.23.1.1, which the ISP already reaches directly.
